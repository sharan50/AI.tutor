## Text identity (§1.3, §5.3)

The script:

```python
#!/usr/bin/env python3
"""Text identity (REBUILD_BRIEF §1.3 and §5.3): for every page served at the
gate commit, the text of <main class="sheet" id="doc"> with tags stripped and
whitespace collapsed, compared with the same page now; index.html at the gate
against contents.html now. Prints every difference as word-level edits."""
import difflib, html, re, subprocess, sys
GATE = sys.argv[1] if len(sys.argv) > 1 else "19cefc4"
def main_text(s):
    m = re.search(r'<main class="sheet" id="doc">(.*?)</main>', s, re.S)
    return re.sub(r"\s+", " ", html.unescape(re.sub(r"<[^>]+>", " ", m.group(1)))).strip()
pages = [p for p in subprocess.run(["git", "ls-tree", "--name-only", "-r", GATE, "docs", "evidence"],
                                   capture_output=True, text=True).stdout.split() if p.endswith(".html")]
bad = 0
for p in pages:
    before = main_text(subprocess.run(["git", "show", f"{GATE}:{p}"], capture_output=True, text=True).stdout)
    now = "docs/contents.html" if p == "docs/index.html" else p
    after = main_text(open(now, encoding="utf-8").read())
    label = p + (" -> " + now if now != p else "")
    if before == after:
        print(f"identical  {label}")
        continue
    a, b = before.split(), after.split()
    ops = [(t, " ".join(a[i1:i2]), " ".join(b[j1:j2]))
           for t, i1, i2, j1, j2 in difflib.SequenceMatcher(None, a, b, autojunk=False).get_opcodes() if t != "equal"]
    print(f"DIFFERS    {label}")
    for t, x, y in ops:
        print(f"    {t:8s} {x[:80]!r} -> {y[:80]!r}" + (f" ({len(y.split())} words)" if len(y.split()) > 12 else ""))
    bad += 1
print(f"{len(pages) - bad} identical, {bad} differ")
```
